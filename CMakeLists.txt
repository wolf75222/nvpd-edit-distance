# Use local release archives on compute nodes that have no internet access.
function(pinned_source name repository revision sha256)
  set(url "https://codeload.github.com/${repository}/tar.gz/${revision}")
  if(EXISTS "${CMAKE_SOURCE_DIR}/deps/${name}.tar.gz")
    set(url "${CMAKE_SOURCE_DIR}/deps/${name}.tar.gz")
  endif()
  FetchContent_Declare(${name} URL "${url}" URL_HASH "SHA256=${sha256}"
                       SOURCE_SUBDIR skip_upstream_executables)
  FetchContent_MakeAvailable(${name})
  set(${name}_SOURCE_DIR "${${name}_SOURCE_DIR}" PARENT_SCOPE)
endfunction()
